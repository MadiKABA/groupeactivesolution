import puppeteer from "puppeteer-core";
const b = await puppeteer.launch({ executablePath: "/usr/bin/google-chrome", headless: true, args: ["--no-sandbox"] });
for (const w of [360, 1440, 1920]) {
  const p = await b.newPage(); await p.setViewport({ width: w, height: 900, isMobile: w < 768 });
  await p.goto("http://localhost:3100/", { waitUntil: "networkidle0" });
  const total = await p.evaluate(() => document.documentElement.scrollHeight);
  for (let y = 0; y < total; y += 500) { await p.evaluate((v) => scrollTo(0, v), y); await new Promise(r => setTimeout(r, 90)); }
  await new Promise(r => setTimeout(r, 2000));
  const rows = await p.evaluate(() => [...document.querySelectorAll("main p, footer p")].filter(el => !el.closest("[hidden],[aria-hidden=true],noscript,.sr-only") && el.getClientRects().length && el.textContent.trim().length > 60).map(el => {
    const cs = getComputedStyle(el); const lh = parseFloat(cs.lineHeight) || parseFloat(cs.fontSize) * 1.5;
    const lines = Math.max(1, Math.round(el.getBoundingClientRect().height / lh));
    return { sec: el.closest("section,footer")?.id || "footer", size: cs.fontSize, lh: (lh / parseFloat(cs.fontSize)).toFixed(2), cpl: Math.round(el.textContent.trim().length / lines), lines, t: el.textContent.trim().slice(0, 40) };
  }));
  console.log("==", w); rows.forEach(r => console.log(`  ${r.sec.padEnd(12)} ${r.size} lh=${r.lh} ~${r.cpl} car./ligne (${r.lines} l.) ${r.t}`));
  await p.close();
}
await b.close();
