// Contraste effectif (couleur du texte composée avec l'opacité, fond composé des ancêtres).
import fs from "node:fs";
import puppeteer from "puppeteer-core";

const browser = await puppeteer.launch({ executablePath: "/usr/bin/google-chrome", headless: true, args: ["--no-sandbox"] });
const out = {};
for (const width of [390, 1440]) {
  const page = await browser.newPage();
  await page.setViewport({ width, height: 900, isMobile: width < 768, hasTouch: width < 768 });
  await page.goto("http://localhost:3100/", { waitUntil: "networkidle0" });
  const total = await page.evaluate(() => document.documentElement.scrollHeight);
  for (let y = 0; y < total; y += 400) {
    await page.evaluate((v) => window.scrollTo(0, v), y);
    await new Promise((r) => setTimeout(r, 110));
  }
  await new Promise((r) => setTimeout(r, 2500));
  out[width] = await page.evaluate(() => {
    const ctx = document.createElement("canvas").getContext("2d");
    const parse = (c) => {
      ctx.fillStyle = "#000";
      ctx.fillStyle = c;
      const v = ctx.fillStyle;
      if (v.startsWith("#")) return [parseInt(v.slice(1, 3), 16), parseInt(v.slice(3, 5), 16), parseInt(v.slice(5, 7), 16), 1];
      const m = v.match(/[\d.]+/g).map(Number);
      return [m[0], m[1], m[2], m[3] ?? 1];
    };
    // Couleurs CSS modernes (lab, oklab, color-mix) : rendues par le navigateur via un élément témoin.
    const probe = document.createElement("div");
    document.body.append(probe);
    const resolve = (c) => {
      probe.style.color = c;
      const s = getComputedStyle(probe).color;
      if (s.startsWith("rgb")) return parse(s);
      // lab()/oklab() : passer par un canvas 1×1
      const cv = document.createElement("canvas");
      cv.width = cv.height = 1;
      const x = cv.getContext("2d");
      x.fillStyle = s;
      x.fillRect(0, 0, 1, 1);
      const d = x.getImageData(0, 0, 1, 1).data;
      const alpha = (s.match(/\/\s*([\d.]+)\)/) || [])[1];
      return [d[0], d[1], d[2], alpha ? Number(alpha) : d[3] / 255];
    };
    const lum = ([r, g, b]) => {
      const f = (v) => ((v /= 255) <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4);
      return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
    };
    const blend = (top, bottom) => {
      const a = top[3];
      return [top[0] * a + bottom[0] * (1 - a), top[1] * a + bottom[1] * (1 - a), top[2] * a + bottom[2] * (1 - a), 1];
    };
    const ratio = (a, b) => {
      const [l1, l2] = [lum(a), lum(b)].sort((x, y) => y - x);
      return (l1 + 0.05) / (l2 + 0.05);
    };
    const rows = new Map();
    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
    let node;
    while ((node = walker.nextNode())) {
      const text = node.textContent.trim();
      if (!text) continue;
      const el = node.parentElement;
      if (!el || el.closest("script,style,noscript,.sr-only,[hidden],[inert],[aria-hidden=true],svg")) continue;
      const cs = getComputedStyle(el);
      const r = el.getBoundingClientRect();
      if (cs.visibility === "hidden" || r.width === 0 || r.height === 0) continue;
      let opacity = 1;
      for (let e = el; e && e !== document.documentElement; e = e.parentElement) opacity *= parseFloat(getComputedStyle(e).opacity);
      if (opacity < 0.05) continue;
      const gradientText = cs.backgroundClip === "text" || cs.webkitBackgroundClip === "text" || (cs.color.includes(", 0)") && cs.backgroundImage !== "none");
      // Fond : composition des background-color des ancêtres, du plus proche au plus lointain jusqu'à un fond opaque.
      const layers = [];
      let image = null;
      for (let e = el; e; e = e.parentElement) {
        const s = getComputedStyle(e);
        if (s.backgroundImage !== "none" && !image && e !== el) image = (e.className?.baseVal ?? e.className ?? "").toString().slice(0, 40);
        const bg = resolve(s.backgroundColor);
        if (bg[3] > 0) layers.push(bg);
        if (bg[3] >= 1) break;
      }
      let bg = [255, 255, 255, 1];
      for (let i = layers.length - 1; i >= 0; i -= 1) bg = blend(layers[i], bg);
      let fg = resolve(cs.color);
      fg = [fg[0], fg[1], fg[2], fg[3] * opacity];
      const fgFlat = blend(fg, bg);
      const size = parseFloat(cs.fontSize);
      const weight = parseInt(cs.fontWeight, 10);
      const large = size >= 24 || (size >= 18.66 && weight >= 700);
      const cr = gradientText ? null : ratio(fgFlat, bg);
      const section = el.closest("section,header,footer,dialog")?.id || el.closest("section,header,footer,dialog")?.tagName.toLowerCase();
      const key = `${section}|${cs.color}|${bg.map(Math.round).join(",")}|${size}|${weight}|${gradientText}|${opacity.toFixed(2)}`;
      const row = rows.get(key) ?? { section, color: cs.color, opacity: +opacity.toFixed(2), bg: `rgb(${bg.map(Math.round).slice(0, 3).join(",")})`, overImage: image, size, weight, large, ratio: cr ? +cr.toFixed(2) : null, gradientText, samples: [] };
      if (row.samples.length < 3) row.samples.push(text.slice(0, 40));
      rows.set(key, row);
    }
    probe.remove();
    return [...rows.values()];
  });
  await page.close();
}
fs.writeFileSync("out/contrast.json", JSON.stringify(out, null, 2));
await browser.close();
for (const [w, rows] of Object.entries(out)) {
  console.log("=====", w, "paires distinctes", rows.length);
  const fails = rows.filter((r) => r.ratio !== null && r.ratio < (r.large ? 3 : 4.5));
  const low = rows.filter((r) => r.ratio !== null && r.ratio >= (r.large ? 3 : 4.5) && r.ratio < (r.large ? 3.5 : 5));
  console.log("ÉCHECS", fails.length);
  fails.forEach((r) => console.log(" ", r.section, r.ratio, `${r.size}px/${r.weight}`, r.color, "op", r.opacity, "sur", r.bg, r.overImage ? `(fond image: ${r.overImage})` : "", JSON.stringify(r.samples)));
  console.log("LIMITES (marge < 0.5)", low.length);
  low.forEach((r) => console.log(" ", r.section, r.ratio, `${r.size}px/${r.weight}`, r.color, "op", r.opacity, "sur", r.bg, r.overImage ? `(fond image: ${r.overImage})` : "", JSON.stringify(r.samples)));
  console.log("TEXTE EN DÉGRADÉ");
  rows.filter((r) => r.gradientText).forEach((r) => console.log(" ", r.section, `${r.size}px/${r.weight}`, "sur", r.bg, JSON.stringify(r.samples)));
}
